#define F_CPU 16000000UL

#include <avr/io.h>
#include <avr/interrupt.h>
#include <avr/eeprom.h>
#include <util/delay.h>
#include <stdint.h>

// Configuracion de pines 

#define DATA_PIN PD6
#define DATA_PORT PORTD
#define DATA_DDR DDRD

#define JOY_SW PD4
#define BUZZER PB3

#define NUM_LEDS 64
#define MAX_NIVEL 32

#define UMBRAL_BAJO 350
#define UMBRAL_ALTO 700

#define ROJO 0
#define VERDE 1
#define AZUL 2
#define AMARILLO 3
#define SIN_COLOR 255

// Variables del juego 

static uint8_t secuencia[MAX_NIVEL];
static uint8_t nivel;
static uint16_t semilla = 0xACE1;

uint16_t EEMEM record_guardado;

// Matriz WS2812B 

static inline void sendByte(uint8_t value)
{
    for (uint8_t mask = 0x80; mask; mask >>= 1)
    {
        if (value & mask)
        {
            DATA_PORT |= (1 << DATA_PIN);

            __asm__ __volatile__(
                "nop\n\t""nop\n\t""nop\n\t""nop\n\t"
                "nop\n\t""nop\n\t""nop\n\t""nop\n\t"
                "nop\n\t""nop\n\t""nop\n\t"
            );

            DATA_PORT &= ~(1 << DATA_PIN);

            __asm__ __volatile__("nop\n\t""nop\n\t");
        }
        else
        {
            DATA_PORT |= (1 << DATA_PIN);

            __asm__ __volatile__("nop\n\t""nop\n\t");

            DATA_PORT &= ~(1 << DATA_PIN);

            __asm__ __volatile__(
                "nop\n\t""nop\n\t""nop\n\t""nop\n\t"
                "nop\n\t""nop\n\t""nop\n\t"
            );
        }
    }
}

static void sendColor(uint8_t r, uint8_t g, uint8_t b)
{
    sendByte(g);
    sendByte(r);
    sendByte(b);
}

static void showColor(uint8_t r, uint8_t g, uint8_t b)
{
    uint8_t oldSREG = SREG;
    cli();

    for (uint8_t i = 0; i < NUM_LEDS; i++)
    {
        sendColor(r, g, b);
    }

    SREG = oldSREG;
    _delay_us(80);
}

static void clearMatrix(void)
{
    showColor(0, 0, 0);
}

static void showColorIndex(uint8_t color)
{
    switch (color)
    {
        case ROJO:
            showColor(35, 0, 0);
            break;

        case VERDE:
            showColor(0, 35, 0);
            break;

        case AZUL:
            showColor(0, 0, 35);
            break;

        case AMARILLO:
            showColor(35, 20, 0);
            break;

        default:
            clearMatrix();
            break;
    }
}

// ADC para el joystick

static void adc_init(void)
{
    ADMUX = (1 << REFS0);

    ADCSRA = (1 << ADEN)
           | (1 << ADPS2)
           | (1 << ADPS1)
           | (1 << ADPS0);
}

static uint16_t adc_read(uint8_t canal)
{
    ADMUX = (1 << REFS0) | (canal & 0x07);

    ADCSRA |= (1 << ADSC);

    while (ADCSRA & (1 << ADSC))
    {
    }

    return ADC;
}

// Lectura de direccion 

static uint8_t joystick_read(void)
{
    uint16_t x = adc_read(0);
    uint16_t y = adc_read(1);

    if (x < UMBRAL_BAJO)
        return ROJO;

    if (x > UMBRAL_ALTO)
        return VERDE;

    if (y < UMBRAL_BAJO)
        return AZUL;

    if (y > UMBRAL_ALTO)
        return AMARILLO;

    return SIN_COLOR;
}

// Pulsador con antirrebote 

static uint8_t button_pressed(void)
{
    if (!(PIND & (1 << JOY_SW)))
    {
        _delay_ms(25);

        if (!(PIND & (1 << JOY_SW)))
        {
            while (!(PIND & (1 << JOY_SW)))
            {
            }

            _delay_ms(25);
            return 1;
        }
    }

    return 0;
}

// Buzzer

static void buzzer_init(void)
{
    DDRB |= (1 << BUZZER);
    PORTB &= ~(1 << BUZZER);
}

static void beep(uint16_t frecuencia, uint16_t duracion_ms)
{
    if (frecuencia == 0)
        return;

    uint32_t ciclos =
        ((uint32_t)frecuencia * duracion_ms) / 1000UL;

    uint32_t medio_periodo_us = 500000UL / frecuencia;

    for (uint32_t i = 0; i < ciclos; i++)
    {
        PORTB |= (1 << BUZZER);

        for (uint16_t j = 0; j < medio_periodo_us; j++)
            _delay_us(1);

        PORTB &= ~(1 << BUZZER);

        for (uint16_t j = 0; j < medio_periodo_us; j++)
            _delay_us(1);
    }

    PORTB &= ~(1 << BUZZER);
}

static void sonido_correcto(void)
{
    beep(1000, 80);
}

static void sonido_error(void)
{
    beep(250, 350);
}

// Generador seudoaleatorio 

static uint16_t random16(void)
{
    uint16_t bit =
        ((semilla >> 0)
        ^ (semilla >> 2)
        ^ (semilla >> 3)
        ^ (semilla >> 5)) & 1;

    semilla = (semilla >> 1) | (bit << 15);

    if (semilla == 0)
        semilla = 0xACE1;

    return semilla;
}

static uint8_t randomColor(void)
{
    return (uint8_t)(random16() % 4);
}

// Espera a que el joystick vuelva al centro 

static void esperar_centro(void)
{
    while (joystick_read() != SIN_COLOR)
    {
        _delay_ms(20);
    }

    _delay_ms(100);
}

// Muestra la secuencia actual 

static void mostrar_secuencia(void)
{
    clearMatrix();
    _delay_ms(400);

    for (uint8_t i = 0; i < nivel; i++)
    {
        showColorIndex(secuencia[i]);
        _delay_ms(550);

        clearMatrix();
        _delay_ms(300);
    }

    _delay_ms(300);
}

// Espera una seleccion y su confirmacion 

static uint8_t esperar_color(void)
{
    uint8_t color;

    esperar_centro();

    while (1)
    {
        color = joystick_read();

        if (color != SIN_COLOR)
        {
            showColorIndex(color);

            if (button_pressed())
            {
                clearMatrix();
                esperar_centro();
                return color;
            }
        }
        else
        {
            clearMatrix();
        }

        _delay_ms(20);
    }
}

// Indicadores de resultado 

static void mostrar_error(void)
{
    showColorIndex(ROJO);
    sonido_error();
    _delay_ms(300);
    clearMatrix();
}

static void mostrar_nivel_superado(void)
{
    showColorIndex(VERDE);
    sonido_correcto();
    clearMatrix();
    _delay_ms(200);
}

/* Inicio de una partida */

static void iniciar_juego(void)
{
    nivel = 1;

    // Genera una nueva secuencia para esta partida. 
    secuencia[0] = randomColor();

    // Los niveles siguientes agregaran un color. 
}

// main

int main(void)
{
    uint16_t record;
    uint16_t puntuacion;

    DATA_DDR |= (1 << DATA_PIN);
    DATA_PORT &= ~(1 << DATA_PIN);

    DDRD &= ~(1 << JOY_SW);
    PORTD |= (1 << JOY_SW);

    DDRC &= ~((1 << PC0) | (1 << PC1));

    adc_init();
    buzzer_init();

    record = eeprom_read_word(&record_guardado);

    if (record == 0xFFFF || record > MAX_NIVEL)
    {
        record = 0;
        eeprom_update_word(&record_guardado, record);
    }

    clearMatrix();

    // Variacion inicial de la semilla. 
    for (uint8_t i = 0; i < 30; i++)
    {
        semilla ^= adc_read(0);
        semilla ^= (adc_read(1) << 1);
        _delay_ms(10);
    }

    while (1)
    {
        iniciar_juego();
        puntuacion = 0;

        while (nivel <= MAX_NIVEL)
        {
            
             //En cada nivel se agrega un solo color.
             // Los colores anteriores se conservan.
             
            if (nivel > 1)
                secuencia[nivel - 1] = randomColor();

            mostrar_secuencia();

            uint8_t correcto = 1;

            for (uint8_t i = 0; i < nivel; i++)
            {
                uint8_t respuesta = esperar_color();

                if (respuesta != secuencia[i])
                {
                    correcto = 0;
                    break;
                }

                sonido_correcto();
            }

            if (!correcto)
            {
                mostrar_error();

                puntuacion = nivel - 1;

                if (puntuacion > record)
                {
                    record = puntuacion;
                    eeprom_update_word(&record_guardado, record);
                }

                break;
            }

            puntuacion = nivel;
            mostrar_nivel_superado();

            nivel++;
            _delay_ms(300);
        }

        
         // Fin de partida.
         // El jugador pulsa el boton para reiniciar.
         
        clearMatrix();

        while (!button_pressed())
        {
            _delay_ms(20);
        }

        esperar_centro();

        // Mezcla la semilla para la proxima partida. 
        semilla ^= adc_read(0);
        semilla ^= (adc_read(1) << 2);
    }
}
