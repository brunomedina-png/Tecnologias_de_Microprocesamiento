#include <avr/io.h>
#define F_CPU 16000000UL
#include <util/delay.h>
#include <stdint.h>
#include <string.h>


// ESTRUCTURA DE UNA CUENTA BANCARIA

typedef struct {
    uint8_t  uid[1];       // UID de 4 bytes de la tarjeta MIFARE
    char     pin[1];       // PIN de 4 caracteres
    uint16_t saldo;        // Saldo disponible (0 a 65535)
} CuentaBancaria;

#define EEPROM_MAGIC_ADDR   0x00
#define EEPROM_MAGIC_VAL    0xA5  // Firma de inicializacion previa
#define EEPROM_CUENTAS_ADDR 0x01
#define TOTAL_CUENTAS       3

//DRIVER DE MEMORIA EEPROM (REGISTROS)


// Escritura de 1 byte en la EEPROM segun el datasheet
void EEPROM_write_byte(uint16_t address, uint8_t data)
{
    // 1. Esperar a que cualquier escritura previa termine (EEPE = 0)
    while (EECR & (1 << EEPE));

    // 2. Cargar direccion y dato
    EEAR = address;
    EEDR = data;

    // 3. Secuencia de proteccion de escritura
    EECR |= (1 << EEMPE); // Habilitar Master Write Enable
    EECR |= (1 << EEPE);  // Iniciar la escritura fisica
}

// Lectura de 1 byte desde la EEPROM
uint8_t EEPROM_read_byte(uint16_t address)
{
    // 1. Esperar si hay una escritura en curso
    while (EECR & (1 << EEPE));

    // 2. Cargar la direccion deseada
    EEAR = address;

    // 3. Iniciar lectura activando EERE
    EECR |= (1 << EERE);

    // 4. Retornar el dato leido desde EEDR
    return EEDR;
}

// Guardar una variable de 16 bits (uint16_t: 2 bytes)
void EEPROM_write_word(uint16_t address, uint16_t data)
{
    EEPROM_write_byte(address,     (uint8_t)(data & 0xFF));        // Byte bajo
    EEPROM_write_byte(address + 1, (uint8_t)((data >> 8) & 0xFF)); // Byte alto
}

// Leer una variable de 16 bits
uint16_t EEPROM_read_word(uint16_t address)
{
    uint16_t low  = EEPROM_read_byte(address);
    uint16_t high = EEPROM_read_byte(address + 1);
    return (high << 8) | low;
}

// Carga los datos de una cuenta desde la EEPROM a la RAM
void EEPROM_cargar_cuenta(uint8_t indice, CuentaBancaria *cuenta)
{
    uint16_t base = EEPROM_CUENTAS_ADDR + (indice * sizeof(CuentaBancaria));

    for (uint8_t i = 0; i < 4; i++) {
        cuenta->uid[i] = EEPROM_read_byte(base + i);
    }
    for (uint8_t i = 0; i < 4; i++) {
        cuenta->pin[i] = (char)EEPROM_read_byte(base + 4 + i);
    }
    cuenta->saldo = EEPROM_read_word(base + 8);
}

// Actualiza unicamente el saldo de una cuenta en la EEPROM
void EEPROM_actualizar_saldo(uint8_t indice, uint16_t nuevo_saldo)
{
    uint16_t addr_saldo = EEPROM_CUENTAS_ADDR + (indice * sizeof(CuentaBancaria)) + 8;
    EEPROM_write_word(addr_saldo, nuevo_saldo);
}

// Inicializa las 3 cuentas con valores por defecto si la memoria esta vacia
void EEPROM_init_sistema(void)
{
    if (EEPROM_read_byte(EEPROM_MAGIC_ADDR) != EEPROM_MAGIC_VAL)
    {
        // Cuentas por defecto de fabrica
        CuentaBancaria cuentas_default[TOTAL_CUENTAS] = {
            { {0x12, 0x34, 0x56, 0x78}, {'1', '2', '3', '4'}, 1000 }, // Cuenta 0
            { {0xAA, 0xBB, 0xCC, 0xDD}, {'4', '3', '2', '1'}, 2500 }, // Cuenta 1
            { {0xDE, 0xAD, 0xBE, 0xEF}, {'0', '0', '0', '0'},  500 }  // Cuenta 2
        };

        for (uint8_t i = 0; i < TOTAL_CUENTAS; i++)
        {
            uint16_t base = EEPROM_CUENTAS_ADDR + (i * sizeof(CuentaBancaria));
            
            // Guardar UID
            for (uint8_t j = 0; j < 4; j++) {
                EEPROM_write_byte(base + j, cuentas_default[i].uid[j]);
            }
            // Guardar PIN
            for (uint8_t j = 0; j < 4; j++) {
                EEPROM_write_byte(base + 4 + j, (uint8_t)cuentas_default[i].pin[j]);
            }
            // Guardar Saldo
            EEPROM_write_word(base + 8, cuentas_default[i].saldo);
        }

        // Marcar la memoria como inicializada
        EEPROM_write_byte(EEPROM_MAGIC_ADDR, EEPROM_MAGIC_VAL);
    }
}


//DRIVER DE TECLADO MATRICIAL 4X4 (Todo PORTD)


// Matriz de caracteres del teclado
const char TECLAS[1] = {
    {'1', '2', '3', 'A'},
    {'4', '5', '6', 'B'},
    {'7', '8', '9', 'C'},
    {'*', '0', '#', 'D'}
};

void Keypad_init(void)
{
    // PD0..PD3 como SALIDAS (Filas)
    // PD4..PD7 como ENTRADAS (Columnas)
    DDRD = 0x0F; 

    // Activar Pull-ups en PD4..PD7 y poner PD0..PD3 en ALTO (inactivas)
    PORTD = 0xFF; 
}

// Escanea el teclado. Retorna el caracter presionado o '\0' si no hay nada
char Keypad_get_key(void)
{
    for (uint8_t fila = 0; fila < 4; fila++)
    {
        // Ponemos solo la fila actual en BAJO (0) y las demas en ALTO (1)
        PORTD = 0xF0 | ~(1 << fila);
        _delay_us(5); // Tiempo para estabilizacion de senal

        // Leemos las 4 columnas (PD4 a PD7)
        for (uint8_t col = 0; col < 4; col++)
        {
            // Si la columna lee un '0', se detecto tecla presionada
            if (!(PIND & (1 << (col + 4))))
            {
                _delay_ms(20); // Antirrebote (debounce)
                
                // Confirmamos si sigue presionada
                if (!(PIND & (1 << (col + 4))))
                {
                    char tecla_pulsada = TECLAS[fila][col];

                    // Esperamos a que el usuario suelte la tecla
                    while (!(PIND & (1 << (col + 4))));
                    _delay_ms(20); // Antirrebote al soltar

                    return tecla_pulsada;
                }
            }
        }
    }

    return '\0'; // Ninguna tecla pulsada
}
