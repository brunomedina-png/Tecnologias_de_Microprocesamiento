#define F_CPU 16000000UL
#include <avr/io.h>
#include <util/delay.h>
#include <stdio.h>
#include <string.h>
#include <stdint.h>
#define LED_VERDE PC0
#define LED_ROJO PC1
#define BUZZER PC2
#define PCF8574_ADDR (0x27 << 1)

// Configura el bus I2C para comunicarse con la pantalla.
void i2c_init(void) {
	TWBR = 72;
	TWSR = 0x00;
}

// Inicia una comunicación I2C.
uint8_t i2c_start(void) {
	uint16_t timeout = 1000;
	TWCR = (1 << TWINT) | (1 << TWSTA) | (1 << TWEN);
	while (!(TWCR & (1 << TWINT))) {
		if (--timeout == 0) return 0;
	}
	return 1;
}

// Finaliza la comunicación I2C.
void i2c_stop(void) {
	TWCR = (1 << TWINT) | (1 << TWSTO) | (1 << TWEN);
	_delay_us(40);
}

// Envía un byte por el bus I2C.
uint8_t i2c_write(uint8_t data) {
	uint16_t timeout = 1000;
	TWDR = data;
	TWCR = (1 << TWINT) | (1 << TWEN);
	while (!(TWCR & (1 << TWINT))) {
		if (--timeout == 0) return 0;
	}
	return 1;
}

// Envía un dato al expansor I2C de la pantalla.
void pcf8574_send(uint8_t val) {
	if (!i2c_start()) return;
	i2c_write(PCF8574_ADDR);
	i2c_write(val | 0x08);
	i2c_stop();
}

// Genera el pulso de habilitación de la pantalla.
void lcd_pulse(uint8_t val) {
	pcf8574_send(val | 0x04);
	_delay_us(2);
	pcf8574_send(val & ~0x04);
	_delay_us(50);
}

// Envía cuatro bits a la pantalla.
void lcd_send_nibble(uint8_t nibble, uint8_t mode) {
	uint8_t data = (nibble & 0xF0) | mode;
	pcf8574_send(data);
	lcd_pulse(data);
}

// Envía una instrucción a la pantalla LCD.
void lcd_cmd(uint8_t cmd) {
	lcd_send_nibble(cmd & 0xF0, 0);
	lcd_send_nibble((cmd << 4) & 0xF0, 0);
	_delay_ms(2);
}

// Envía un carácter para mostrarlo en la pantalla.
void lcd_char(char c) {
	lcd_send_nibble(((uint8_t)c) & 0xF0, 1);
	lcd_send_nibble((((uint8_t)c) << 4) & 0xF0, 1);
	_delay_us(50);
}

// Muestra una cadena de texto en la pantalla.
void lcd_msg(const char *s) {
	while (*s) {
		lcd_char(*s++);
	}
}

// Ubica el cursor en la fila y columna indicadas.
void lcd_set_cursor(uint8_t row, uint8_t col) {
	uint8_t addr = (row == 0) ? (0x80 + col) : (0xC0 + col);
	lcd_cmd(addr);
}

// Borra el contenido de la pantalla.
void lcd_clear(void) {
	lcd_cmd(0x01);
	_delay_ms(2);
}

// Inicializa la pantalla LCD en modo de cuatro bits.
void lcd_init(void) {
	i2c_init();
	_delay_ms(50);

	lcd_send_nibble(0x30, 0); _delay_ms(5);
	lcd_send_nibble(0x30, 0); _delay_us(150);
	lcd_send_nibble(0x30, 0); _delay_us(150);
	lcd_send_nibble(0x20, 0); _delay_us(150);

	lcd_cmd(0x28);
	lcd_cmd(0x0C);
	lcd_cmd(0x06);
	lcd_clear();
}


// Primera etapa: prueba del LCD, teclado y acceso basico.

// Distribución de teclas usada en el teclado de Proteus.
const char TECLAS[4][4] = {
	{'7', '8', '9', '/'},
	{'4', '5', '6', 'X'},
	{'1', '2', '3', '-'},
	{'O', '0', '=', '+'}
};

// Configura las filas y columnas del teclado.
void Keypad_init(void) {
	DDRD = 0x0F;
	PORTD = 0xFF;
}

// Lee el teclado y devuelve la tecla presionada.
char Keypad_get_key(void) {
	for (uint8_t fila = 0; fila < 4; fila++) {
		PORTD = 0xF0 | ~(1 << fila);
		_delay_us(5);

		for (uint8_t col = 0; col < 4; col++) {
			if (!(PIND & (1 << (col + 4)))) {
				_delay_ms(20);
				if (!(PIND & (1 << (col + 4)))) {
					char c = TECLAS[fila][col];
					while (!(PIND & (1 << (col + 4))));
					_delay_ms(20);
					PORTD = 0xFF;
					return c;
				}
			}
		}
	}
	PORTD = 0xFF;
	return '\0';
}

// Enciende el LED verde para indicar una operación correcta.
void feedback_exito(void) {
	PORTC |= (1 << LED_VERDE);
	_delay_ms(150);
	PORTC &= ~(1 << LED_VERDE);
}

// Enciende el LED rojo para indicar un error.
void feedback_error(void) {
	PORTC |= (1 << LED_ROJO);
	_delay_ms(300);
	PORTC &= ~(1 << LED_ROJO);
}

// Enciende el LED verde para indicar una operación correcta.
void feedback_exito(void) {
	PORTC |= (1 << LED_VERDE);
	_delay_ms(150);
	PORTC &= ~(1 << LED_VERDE);
}

// Enciende el LED rojo para indicar un error.
void feedback_error(void) {
	PORTC |= (1 << LED_ROJO);
	_delay_ms(300);
	PORTC &= ~(1 << LED_ROJO);
}



int main(void) {
    DDRC |= (1 << LED_VERDE) | (1 << LED_ROJO) | (1 << BUZZER);
    PORTC &= ~((1 << LED_VERDE) | (1 << LED_ROJO) | (1 << BUZZER));
    Keypad_init();
    lcd_init();

    char pin[5] = {0};
    uint8_t pos = 0;
    int8_t cuenta = -1;
    const char *pines[3] = {"1234", "1111", "0000"};
    char tecla;

    while (1) {
        if (cuenta < 0) {
            lcd_set_cursor(0, 0); lcd_msg("Seleccione cuenta");
            lcd_set_cursor(1, 0); lcd_msg("/ A  X B  - C    ");
            tecla = Keypad_get_key();
            if (tecla == '/') cuenta = 0;
            else if (tecla == 'X') cuenta = 1;
            else if (tecla == '-') cuenta = 2;
            if (cuenta >= 0) { pos = 0; memset(pin, 0, sizeof(pin)); lcd_clear(); }
        } else {
            lcd_set_cursor(0, 0); lcd_msg("Ingrese PIN:     ");
            lcd_set_cursor(1, 0);
            for (uint8_t i = 0; i < 4; i++) lcd_char(i < pos ? '*' : ' ');
            lcd_msg("            ");
            tecla = Keypad_get_key();
            if (tecla >= '0' && tecla <= '9' && pos < 4) pin[pos++] = tecla;
            else if (tecla == 'O' && pos > 0) pin[--pos] = '\0';
            if (pos == 4) {
                if (strncmp(pin, pines[(uint8_t)cuenta], 4) == 0) {
                    lcd_clear(); lcd_set_cursor(0,0); lcd_msg("Acceso correcto");
                    feedback_exito(); _delay_ms(700);
                } else {
                    lcd_clear(); lcd_set_cursor(0,0); lcd_msg("PIN incorrecto");
                    feedback_error(); _delay_ms(700);
                }
                cuenta = -1; pos = 0; memset(pin, 0, sizeof(pin)); lcd_clear();
            }
        }
    }
}
