#include <stdio.h>

#define LED_VERDE PC0
#define LED_ROJO PC1
#define BUZZER PC2
#define PCF8574_ADDR (0x27 << 1)

// Configura el bus I2C para comunicarse con la pantalla.
void i2c_init(void) {
	TWBR = 72;
	TWSR = 0x00;
}

// Inicia una comunicación I2C.
uint8_t i2c_start(void) {
	uint16_t timeout = 1000;
	TWCR = (1 << TWINT) | (1 << TWSTA) | (1 << TWEN);
	while (!(TWCR & (1 << TWINT))) {
		if (--timeout == 0) return 0;
	}
	return 1;
}

// Finaliza la comunicación I2C.
void i2c_stop(void) {
	TWCR = (1 << TWINT) | (1 << TWSTO) | (1 << TWEN);
	_delay_us(40);
}

// Envía un byte por el bus I2C.
uint8_t i2c_write(uint8_t data) {
	uint16_t timeout = 1000;
	TWDR = data;
	TWCR = (1 << TWINT) | (1 << TWEN);
	while (!(TWCR & (1 << TWINT))) {
		if (--timeout == 0) return 0;
	}
	return 1;
}

// Envía un dato al expansor I2C de la pantalla.
void pcf8574_send(uint8_t val) {
	if (!i2c_start()) return;
	i2c_write(PCF8574_ADDR);
	i2c_write(val | 0x08);
	i2c_stop();
}

// Genera el pulso de habilitación de la pantalla.
void lcd_pulse(uint8_t val) {
	pcf8574_send(val | 0x04);
	_delay_us(2);
	pcf8574_send(val & ~0x04);
	_delay_us(50);
}

// Envía cuatro bits a la pantalla.
void lcd_send_nibble(uint8_t nibble, uint8_t mode) {
	uint8_t data = (nibble & 0xF0) | mode;
	pcf8574_send(data);
	lcd_pulse(data);
}

// Envía una instrucción a la pantalla LCD.
void lcd_cmd(uint8_t cmd) {
	lcd_send_nibble(cmd & 0xF0, 0);
	lcd_send_nibble((cmd << 4) & 0xF0, 0);
	_delay_ms(2);
}

// Envía un carácter para mostrarlo en la pantalla.
void lcd_char(char c) {
	lcd_send_nibble(((uint8_t)c) & 0xF0, 1);
	lcd_send_nibble((((uint8_t)c) << 4) & 0xF0, 1);
	_delay_us(50);
}

// Muestra una cadena de texto en la pantalla.
void lcd_msg(const char *s) {
	while (*s) {
		lcd_char(*s++);
	}
}

// Ubica el cursor en la fila y columna indicadas.
void lcd_set_cursor(uint8_t row, uint8_t col) {
	uint8_t addr = (row == 0) ? (0x80 + col) : (0xC0 + col);
	lcd_cmd(addr);
}

// Borra el contenido de la pantalla.
void lcd_clear(void) {
	lcd_cmd(0x01);
	_delay_ms(2);
}

// Inicializa la pantalla LCD en modo de cuatro bits.
void lcd_init(void) {
	i2c_init();
	_delay_ms(50);

	lcd_send_nibble(0x30, 0); _delay_ms(5);
	lcd_send_nibble(0x30, 0); _delay_us(150);
	lcd_send_nibble(0x30, 0); _delay_us(150);
	lcd_send_nibble(0x20, 0); _delay_us(150);

	lcd_cmd(0x28);
	lcd_cmd(0x0C);
	lcd_cmd(0x06);
	lcd_clear();
}


typedef struct { uint8_t uid[4]; char pin[4]; uint16_t saldo; } CuentaBancaria;
#define TOTAL_CUENTAS 3
#define EEPROM_MAGIC_ADDR 0x00
#define EEPROM_MAGIC_VAL 0xC5
#define EEPROM_BASE_ADDR 0x02
#define CUENTA_BYTES 10

// Guarda un byte en la EEPROM.
void EEPROM_write_byte(uint16_t addr, uint8_t data) {
	while (EECR & (1 << EEPE));
	EEAR = addr;
	EEDR = data;
	EECR |= (1 << EEMPE);
	EECR |= (1 << EEPE);
}

// Lee un byte de la EEPROM.
uint8_t EEPROM_read_byte(uint16_t addr) {
	while (EECR & (1 << EEPE));
	EEAR = addr;
	EECR |= (1 << EERE);
	return EEDR;
}

// Guarda un número de dos bytes en la EEPROM.
void EEPROM_write_word(uint16_t addr, uint16_t data) {
	EEPROM_write_byte(addr,     (uint8_t)(data & 0xFF));
	EEPROM_write_byte(addr + 1, (uint8_t)((data >> 8) & 0xFF));
}

// Lee un número de dos bytes de la EEPROM.
uint16_t EEPROM_read_word(uint16_t addr) {
	uint16_t l = EEPROM_read_byte(addr);
	uint16_t h = EEPROM_read_byte(addr + 1);
	return (h << 8) | l;
}

// Carga los datos de una cuenta desde la EEPROM.
void EEPROM_cargar_cuenta(uint8_t idx, CuentaBancaria *c) {
	uint16_t base = EEPROM_BASE_ADDR + ((uint16_t)idx * CUENTA_BYTES);
	for (uint8_t i = 0; i < 4; i++) {
		c->uid[i] = EEPROM_read_byte(base + i);
	}
	for (uint8_t i = 0; i < 4; i++) {
		c->pin[i] = (char)EEPROM_read_byte(base + 4 + i);
	}
	c->saldo = EEPROM_read_word(base + 8);
}

// Guarda los datos de una cuenta en la EEPROM.
void EEPROM_guardar_cuenta(uint8_t idx, const CuentaBancaria *c) {
	uint16_t base = EEPROM_BASE_ADDR + ((uint16_t)idx * CUENTA_BYTES);
	for (uint8_t i = 0; i < 4; i++) {
		EEPROM_write_byte(base + i, c->uid[i]);
	}
	for (uint8_t i = 0; i < 4; i++) {
		EEPROM_write_byte(base + 4 + i, (uint8_t)c->pin[i]);
	}
	EEPROM_write_word(base + 8, c->saldo);
}

// Carga las cuentas iniciales si la EEPROM no está inicializada.
void EEPROM_init_sistema(void) {
	if (EEPROM_read_byte(EEPROM_MAGIC_ADDR) != EEPROM_MAGIC_VAL) {
		CuentaBancaria c_def[TOTAL_CUENTAS] = {
			{{0,0,0,0}, {'1','2','3','4'}, 1000},
			{{0,0,0,0}, {'1','1','1','1'}, 2500},
			{{0,0,0,0}, {'0','0','0','0'},  500}
		};

		for (uint8_t i = 0; i < TOTAL_CUENTAS; i++) {
			EEPROM_guardar_cuenta(i, &c_def[i]);
		}
		EEPROM_write_byte(EEPROM_MAGIC_ADDR, EEPROM_MAGIC_VAL);
	}
}

// Distribución de teclas usada en el teclado de Proteus.
const char TECLAS[4][4] = {
	{'7', '8', '9', '/'},
	{'4', '5', '6', 'X'},
	{'1', '2', '3', '-'},
	{'O', '0', '=', '+'}
};

// Configura las filas y columnas del teclado.
void Keypad_init(void) {
	DDRD = 0x0F;
	PORTD = 0xFF;
}

// Lee el teclado y devuelve la tecla presionada.
char Keypad_get_key(void) {
	for (uint8_t fila = 0; fila < 4; fila++) {
		PORTD = 0xF0 | ~(1 << fila);
		_delay_us(5);

		for (uint8_t col = 0; col < 4; col++) {
			if (!(PIND & (1 << (col + 4)))) {
				_delay_ms(20);
				if (!(PIND & (1 << (col + 4)))) {
					char c = TECLAS[fila][col];
					while (!(PIND & (1 << (col + 4))));
					_delay_ms(20);
					PORTD = 0xFF;
					return c;
				}
			}
		}
	}
	PORTD = 0xFF;
	return '\0';
}

// Enciende el LED verde para indicar una operación correcta.
void feedback_exito(void) {
	PORTC |= (1 << LED_VERDE);
	_delay_ms(150);
	PORTC &= ~(1 << LED_VERDE);
}

// Enciende el LED rojo para indicar un error.
void feedback_error(void) {
	PORTC |= (1 << LED_ROJO);
	_delay_ms(300);
	PORTC &= ~(1 << LED_ROJO);
}

// Enciende el LED verde para indicar una operación correcta.
void feedback_exito(void) {
	PORTC |= (1 << LED_VERDE);
	_delay_ms(150);
	PORTC &= ~(1 << LED_VERDE);
}

// Enciende el LED rojo para indicar un error.
void feedback_error(void) {
	PORTC |= (1 << LED_ROJO);
	_delay_ms(300);
	PORTC &= ~(1 << LED_ROJO);
}



int main(void) {
    DDRC |= (1 << LED_VERDE) | (1 << LED_ROJO) | (1 << BUZZER);
    PORTC &= ~((1 << LED_VERDE) | (1 << LED_ROJO) | (1 << BUZZER));
    Keypad_init(); lcd_init(); EEPROM_init_sistema();
    char pin_ingresado[5] = {0};
    uint8_t pin_pos = 0;
    char buffer[17];
    int8_t cuenta_activa = -1;
    uint16_t monto = 0;
    uint8_t autenticado = 0;

    while (1) {
        if (cuenta_activa < 0) {
            lcd_set_cursor(0,0); lcd_msg("Acerque Tarjeta  ");
            lcd_set_cursor(1,0); lcd_msg("/ Cuenta1 X2 -3  ");
            char t = Keypad_get_key();
            if (t == '/') cuenta_activa = 0;
            else if (t == 'X') cuenta_activa = 1;
            else if (t == '-') cuenta_activa = 2;
            if (cuenta_activa >= 0) { pin_pos=0; memset(pin_ingresado,0,sizeof(pin_ingresado)); autenticado=0; lcd_clear(); }
        } else if (!autenticado) {
            lcd_set_cursor(0,0); lcd_msg("Ingrese PIN:     ");
            lcd_set_cursor(1,0);
            for (uint8_t i=0;i<4;i++) lcd_char(i<pin_pos?'*':' ');
            lcd_msg("            ");
            char t=Keypad_get_key();
            if (t>='0' && t<='9' && pin_pos<4) pin_ingresado[pin_pos++]=t;
            else if (t=='O' && pin_pos>0) pin_ingresado[--pin_pos]='\0';
            if (pin_pos==4) {
                CuentaBancaria c; EEPROM_cargar_cuenta((uint8_t)cuenta_activa,&c);
                if (strncmp(pin_ingresado,c.pin,4)==0) { autenticado=1; feedback_exito(); lcd_clear(); }
                else { lcd_clear(); lcd_set_cursor(0,0); lcd_msg("PIN incorrecto"); feedback_error(); _delay_ms(600); pin_pos=0; memset(pin_ingresado,0,sizeof(pin_ingresado)); lcd_clear(); }
            }
        } else {
            lcd_set_cursor(0,0); lcd_msg("1:Saldo 2:Retiro");
            lcd_set_cursor(1,0); lcd_msg("3:Deposito +:Salir");
            char op=Keypad_get_key();
            if (op=='1') {
                CuentaBancaria c; EEPROM_cargar_cuenta((uint8_t)cuenta_activa,&c);
                lcd_clear(); lcd_set_cursor(0,0); snprintf(buffer,sizeof(buffer),"Cuenta %d",cuenta_activa+1); lcd_msg(buffer);
                lcd_set_cursor(1,0); snprintf(buffer,sizeof(buffer),"Saldo: $%u",c.saldo); lcd_msg(buffer);
                while (Keypad_get_key()!='O') { }
                lcd_clear();
            } else if (op=='2' || op=='3') {
                uint8_t deposito=(op=='3'); monto=0; lcd_clear();
                while (1) {
                    lcd_set_cursor(0,0); lcd_msg(deposito?"Monto deposito  ":"Monto retiro    ");
                    lcd_set_cursor(1,0); snprintf(buffer,sizeof(buffer),"%u               ",monto); lcd_msg(buffer);
                    char t=Keypad_get_key();
                    if (t>='0' && t<='9' && monto<6000) monto=(uint16_t)(monto*10+(t-'0'));
                    else if (t=='O' || t=='*') { monto=0; break; }
                    else if (t=='=') {
                        CuentaBancaria c; EEPROM_cargar_cuenta((uint8_t)cuenta_activa,&c);
                        if (monto>0 && ((deposito && monto <= (uint16_t)(65535U-c.saldo)) || (!deposito && monto<=c.saldo))) {
                            if (deposito) c.saldo+=monto; else c.saldo-=monto;
                            EEPROM_guardar_cuenta((uint8_t)cuenta_activa,&c);
                            lcd_clear(); lcd_set_cursor(0,0); lcd_msg("Operacion correcta"); feedback_exito();
                        } else { lcd_clear(); lcd_set_cursor(0,0); lcd_msg("Monto no valido"); feedback_error(); }
                        _delay_ms(700); lcd_clear(); break;
                    }
                }
            } else if (op=='+' || op=='O') { cuenta_activa=-1; autenticado=0; lcd_clear(); }
        }
    }
}
