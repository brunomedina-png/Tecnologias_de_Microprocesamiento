; Problema B 
.include "m328pdef.inc"

.org 0x0000
    rjmp INICIO

INICIO:
    ; Configurar Stack
    ldi r16, LOW(RAMEND)
    out SPL, r16
    ldi r16, HIGH(RAMEND)
    out SPH, r16

    ; Puerto B: Entradas con pull-up (PB0, PB1, PB2)
    clr r16
    out DDRB, r16
    ldi r16, 0x07
    out PORTB, r16

    ; Puerto D: Salidas para el display
    ser r16
    out DDRD, r16

    clr r17 ; r17 guardara el contador (empieza en 0)
BUCLE_PRINCIPAL:
    rcall MOSTRAR_NUMERO

    ; Chequear Incremento (PB0)
    sbis PINB, 0
    rjmp SUMAR

    ; Chequear Decremento (PB1)
    sbis PINB, 1
    rjmp RESTAR

    ; Chequear Reset (PB2)
    sbis PINB, 2
    rjmp REINICIAR

    rjmp BUCLE_PRINCIPAL

SUMAR:
    rcall ESPERAR_BOTON
    cpi r17, 9
    breq FIN_SUMAR
    inc r17
FIN_SUMAR:
    rjmp BUCLE_PRINCIPAL

RESTAR:
    rcall ESPERAR_BOTON
    cpi r17, 0
    breq FIN_RESTAR
    dec r17
FIN_RESTAR:
    rjmp BUCLE_PRINCIPAL

REINICIAR:
    rcall ESPERAR_BOTON
    clr r17
    rjmp BUCLE_PRINCIPAL
MOSTRAR_NUMERO:
    ; Cargar direccion de la tabla en el registro Z
    ldi ZH, HIGH(TABLA_7SEG * 2)
    ldi ZL, LOW(TABLA_7SEG * 2)

    ; Sumar la posicion actual del contador (r17)
    add ZL, r17
    clr r16
    adc ZH, r16

    ; Leer el valor de la memoria Flash y enviarlo al Puerto D
    lpm r16, Z
    out PORTD, r16
    ret

; Patrones hexadecimales de 0 a 9 para catodo comun
TABLA_7SEG:
    .db 0x3F, 0x06, 0x5B, 0x4F, 0x66, 0x6D, 0x7D, 0x07, 0x7F, 0x6F

ESPERAR_BOTON:
    rcall DELAY
    ; Espera a que se suelte el boton para evitar conteo continuo
ESPERAR_SOLTAR:
    in r16, PINB
    andi r16, 0x07
    cpi r16, 0x07
    brne ESPERAR_SOLTAR
    rcall DELAY
    ret

; Rutina simple de retardo
DELAY:
    ldi r18, 50
D1: ldi r19, 200
D2: ldi r20, 200
D3: dec r20
    brne D3
    dec r19
    brne D2
    dec r18
    brne D1
    ret
